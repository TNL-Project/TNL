if(TNL_BUILD_CUDA)
    add_executable(tnl-benchmark-gnn-cuda tnl-benchmark-gnn.cu)
    target_link_libraries(tnl-benchmark-gnn-cuda PUBLIC TNL::TNL)
    find_package(CUDAToolkit REQUIRED)
    target_link_libraries(tnl-benchmark-gnn-cuda PUBLIC CUDA::cublas)
    target_compile_definitions(tnl-benchmark-gnn-cuda PUBLIC HAVE_CUBLAS)
    install(TARGETS tnl-benchmark-gnn-cuda RUNTIME DESTINATION bin COMPONENT benchmarks)
endif()
