if(TNL_BUILD_CUDA)
    add_executable(tnl-benchmark-spmm-cuda tnl-benchmark-spmm.cu)
    target_link_libraries(tnl-benchmark-spmm-cuda PUBLIC TNL::TNL)
    find_package(CUDAToolkit REQUIRED)
    target_link_libraries(tnl-benchmark-spmm-cuda PUBLIC CUDA::cusparse)
    target_compile_definitions(tnl-benchmark-spmm-cuda PUBLIC HAVE_CUSPARSE)
    install(TARGETS tnl-benchmark-spmm-cuda RUNTIME DESTINATION bin COMPONENT benchmarks)
endif()

if(TNL_BUILD_HIP)
    add_executable(tnl-benchmark-spmm-hip tnl-benchmark-spmm.hip)
    target_link_libraries(tnl-benchmark-spmm-hip PUBLIC TNL::TNL)
    find_package(hipsparse)
    if(hipsparse_FOUND)
        target_link_libraries(tnl-benchmark-spmm-hip PUBLIC roc::hipsparse)
        target_compile_definitions(tnl-benchmark-spmm-hip PUBLIC HAVE_HIPSPARSE)
    else()
        message(WARNING "hipSPARSE not found, tnl-benchmark-spmm-hip will run without it.")
    endif()
    install(TARGETS tnl-benchmark-spmm-hip RUNTIME DESTINATION bin COMPONENT benchmarks)
endif()
