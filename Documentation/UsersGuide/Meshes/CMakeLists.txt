set(CPP_EXAMPLES ReadMeshExample MeshConfigurationExample MeshIterationExample GameOfLife)
set(GPU_EXAMPLES ParallelIterationCuda GameOfLifeCuda)
set(ALL_EXAMPLES ${CPP_EXAMPLES})

foreach(target IN ITEMS ${CPP_EXAMPLES})
    tnl_add_example_with_output(${target} SOURCES ${target}.cpp BACKENDS HOST)
endforeach()

foreach(target IN ITEMS ${GPU_EXAMPLES})
    if(target STREQUAL "GameOfLifeCuda")
        set(source GameOfLife.cpp)
    else()
        set(source ${target}.h)
    endif()
    tnl_add_example_with_output(${target} SOURCES ${source} BACKENDS CUDA HIP)
    # the target is not created when neither CUDA nor HIP is enabled
    if(TARGET ${target})
        list(APPEND ALL_EXAMPLES ${target})
    endif()
endforeach()

find_package(ZLIB)
if(ZLIB_FOUND)
    foreach(target IN ITEMS ${ALL_EXAMPLES})
        target_compile_definitions(${target} PUBLIC "-DHAVE_ZLIB")
        target_link_libraries(${target} PUBLIC ZLIB::ZLIB)
    endforeach()
endif()

find_package(tinyxml2)
if(tinyxml2_FOUND)
    foreach(target IN ITEMS ${ALL_EXAMPLES})
        target_compile_definitions(${target} PUBLIC "-DHAVE_TINYXML2")
        target_link_libraries(${target} PUBLIC tinyxml2::tinyxml2)
    endforeach()
endif()

set(DATA_FILES example-triangles.vtu grid-100x100.vtu)

foreach(file IN ITEMS ${DATA_FILES})
    configure_file(${file} ${CMAKE_CURRENT_BINARY_DIR}/${file} COPYONLY)
endforeach()
